To eliminate database bottlenecks during morning synchronization, read queries are served via in-memory Redis clusters with automated DocType invalidation hooks:

# Redis Cache Engine with Single-Flight Mutex (Anti-Thundering Herd) def get_or_set_cached(cache_key, query_fn, ttl=1800): raw = redis.get(cache_key) if raw: return json.loads(raw)["data"] # Acquire distributed mutex so only 1 worker queries MariaDB if redis.set(f"lock:{cache_key}", "locked", nx=True, ex=15): fresh_data = query_fn() redis.setex(cache_key, ttl, json.dumps({"_cached_at": time.time(), "data": fresh_data})) redis.delete(f"lock:{cache_key}") return fresh_data else: # Concurrent request receives cached/grace data without DB hammer return get_grace_cached(cache_key)
